    async function loadJoke() {
  showLoading();                       // "Loading…" + spinner
  try {
    const res = await fetch(API_URL);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);  // 404/500 do not throw by themselves
    const data = await res.json();
    showJoke(data.setup, data.punchline);
  } catch (err) {
    showError(err);                    // friendly message
  } finally {
    hideLoading();                     // always runs
  }
}
